cmake_minimum_required(VERSION 3.19)
project(ModernBMDGeometryTests LANGUAGES CXX)
enable_testing()
add_executable(ModernBMDGeometryTests ModernBMDGeometryTests.cpp)
target_compile_features(ModernBMDGeometryTests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(ModernBMDGeometryTests PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(ModernBMDGeometryTests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME ModernBMDGeometry COMMAND ModernBMDGeometryTests)
